# Pinned, not >= : sinabs>=3.1.3 (below) requires nir<=1.0.4 on its own end, and
# every nirtorch release >=2.0.3 requires nir>=1.0.6 -- those two floors/ceilings
# can never both be satisfied, so a bare >= here lets pip's resolver wander into
# an unresolvable combination. nir==1.0.4 + nirtorch==2.0.2 is the last pairing
# confirmed actually working against sinabs in this project (see
# docs/frameworks/realtime_nir_evaluation.md's feasibility checks, run against
# exactly these versions).
nir==1.0.4

# Deep learning (CPU or CUDA — install the correct wheel for your GPU)
torch>=2.1.0   # install via https://pytorch.org

# torch.compile's inductor backend needs Triton for CUDA kernel codegen.
# Linux: the official `triton` package — pip's own manylinux wheels, Python
# 3.10-3.14, no Windows build exists for it at all (verified against PyPI's
# published wheel list). It normally rides in as a transitive dependency of
# the CUDA torch wheel anyway; listed explicitly here so it's a visible,
# pinned requirement rather than an implicit side effect of installing torch.
# Windows: `triton` has no wheel to install there full stop, at any Python
# version — triton-windows is the actively-maintained fork with real
# prebuilt Windows wheels (confirmed working here: Python 3.14, triton-windows
# 3.7.1.post27). The two markers are mutually exclusive, so exactly one of
# these two lines ever actually installs on a given machine.
triton; sys_platform != "win32"
triton-windows; sys_platform == "win32"
